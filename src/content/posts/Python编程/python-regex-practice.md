正确密码的验证函数 def is_valid_password(password):     # 1. 检查是否包含空白字符（空格、制表符、换行等）     if re.search(r"\s", password):         return False      # 2. 必须同时包含字母、数字、特殊字符     has_digit = re.search(r"\d", password)     has_alpha = re.search(r"[a-zA-Z]", password)     has_special = re.search(r"[!@#$%^&*.?]", password)      return bool(has_digit and has_alpha and has_special)  #
